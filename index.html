const Alexa = require('ask-sdk-core');

const ProjecaoInvestimentoIntentHandler = {
    canHandle(handlerInput) {
        return Alexa.getRequestType(handlerInput.requestEnvelope) === 'IntentRequest'
            && Alexa.getIntentName(handlerInput.requestEnvelope) === 'ProjecaoInvestimentoIntent';
    },
    async handle(handlerInput) {
        // Exemplo de integração com API backend do FinControl Pro
        const mes = Alexa.getSlotValue(handlerInput.requestEnvelope, 'mes') || 'Junho';
        const resposta = `Com aportes de 1000 reais a 120% do CDI, sua projeção para ${mes} é de aproximadamente 10 mil e 528 reais líquidos.`;
        
        return handlerInput.responseBuilder
            .speak(resposta)
            .reprompt('Deseja consultar outro período?')
            .getResponse();
    }
};

Comandos Sugeridos para o Gemini

A API do Gemini permite análises contextuais em tempo real enviando a estrutura do seu aplicativo como um JSON no prompt do sistema. 
// Exemplo de integração usando Google GenAI SDK (JavaScript)
import { GoogleGenAI } from "@google/genai";

const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });

async function analisarFinancas(dadosFinanceiros, perguntaUsuario) {
    const systemInstruction = `Você é um consultor financeiro pessoal especialista em investimentos no Brasil (CDI, Tesouro, LCI/LCA) e quitação de dívidas. 
    Analise a estrutura JSON recebida e dê respostas diretas, motivadoras e numéricas.`;

    const response = await ai.models.generateContent({
        model: 'gemini-1.5-flash',
        contents: [
            { role: 'user', parts: [{ text: `Dados Financeiros: ${JSON.stringify(dadosFinanceiros)}\nPergunta: ${perguntaUsuario}` }] }
        ],
        config: { systemInstruction }
    });

    return response.text;
}
# Exemplo de Script Python com o novo SDK google-genai
from google import genai
from google.genai import types

client = genai.Client()

def gerar_diagnostico_gemini(renda, contas, dividas, aportes):
    prompt = f"""
    Atue como um planejador financeiro CFP®.
    Renda Mensal: R$ {renda}
    Contas Fixas: R$ {contas}
    Dívidas Totais: R$ {dividas}
    Aporte Mensal Planejado: R$ {aportes}
    
    Forneça 3 ações prioritárias para os próximos 6 meses.
    """
    
    response = client.models.generate_content(
        model='gemini-1.5-pro',
        contents=prompt,
        config=types.GenerateContentConfig(
            temperature=0.2,
        )
    )
    
    return response.text
